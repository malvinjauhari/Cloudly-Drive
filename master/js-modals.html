// Modal Workers function openWorkerModal() { document.getElementById('workerModal').classList.remove('hidden'); } function closeWorkerModal() { document.getElementById('workerModal').classList.add('hidden'); } function saveWorker() { const btn = document.getElementById('btnSaveWorker'); btn.innerText = 'Menyimpan...'; const data = { id: document.getElementById('workerId').value, name: document.getElementById('workerName').value, email: document.getElementById('workerEmail').value, url: document.getElementById('workerUrl').value, status: document.getElementById('workerStatus').value }; google.script.run.withSuccessHandler(res => { btn.innerText = 'Simpan'; showToast(res.message, res.status === 'error'); if (res.status === 'success' || res.status === 'warning') { closeWorkerModal(); fetchSettings(); fetchData(); } }).saveWorker(null, data); } function deleteWorker(id) { openConfirmModal('Hapus Worker', "Hapus konfigurasi worker " + id + "?", 'Hapus', 'bg-red-600', () => { google.script.run.withSuccessHandler(res => { showToast(res.message); fetchSettings(); fetchData(); }).deleteWorker(id); }); }
